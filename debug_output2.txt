async function analyzeEvidenceFile(file) {

    if (!file) {
        throw new Error(
            "No evidence file selected."
        );
    }

    const controller =
        new AbortController();

    const timeoutId =
        setTimeout(
            () => controller.abort(),
            API_CONFIG.timeout
        );

    try {

        const formData =
            new FormData();

        formData.append(
            "file",
            file
        );

        console.log(
            "Anvaya -> uploading:",
            file.name
        );

        const response =
            await fetch(
                endpointUrl(
                    API_CONFIG.endpoints.analyze
                ),
                {
                    method: "POST",
                    body: formData,
                    signal: controller.signal
                }
            );

        if (!response.ok) {

            const errorText =
                await response.text();

            throw new Error(
                `Backend returned HTTP ${response.status}: ${errorText}`
            );
        }

        return await response.json();

    } catch (error) {

        if (
            error.name ===
            "AbortError"
        ) {

            throw new Error(
                "Analysis timed out after 120 seconds."
            );
        }

        throw error;

    } finally {

        clearTimeout(
            timeoutId
        );
    }
}
